cmake_minimum_required(VERSION 3.12)
project(ioswitch_on_board CXX)
set(CMAKE_CXX_STANDARD 11)
get_filename_component(DEFAULT_PLATFORM "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(XUANWU_PLATFORM "${DEFAULT_PLATFORM}" CACHE PATH "Xuanwu platform installation")
set(NPC_TEST "${XUANWU_PLATFORM}/npc_test")
set(BIN "${CMAKE_CURRENT_SOURCE_DIR}/../defaultGroup/bin")
set(SDK "${NPC_TEST}/on_board/npc_lib")
execute_process(COMMAND python3 "${NPC_TEST}/tools/autogen_tbl/autogen_table_llt.py" "${BIN}" RESULT_VARIABLE gen_result)
if(NOT gen_result EQUAL 0)
 message(FATAL_ERROR "table header generation failed")
endif()
configure_file("${BIN}/bin_tables.h" "${CMAKE_CURRENT_BINARY_DIR}/IOSwitch_tables.h" COPYONLY)
add_executable(ioswitch_on_board main.cpp table_config.cpp debug_process.cpp "${NPC_TEST}/src/ndi_mgr/ndi_mgr.cpp")
target_include_directories(ioswitch_on_board PRIVATE "${SDK}/include" "${NPC_TEST}/on_board/include" "${NPC_TEST}/src/ndi_mgr" "${NPC_TEST}/3rd_party/nlohmann-json-develop/include/nlohmann" "${NPC_TEST}/3rd_party/nlohmann-json-develop/include" "${CMAKE_CURRENT_BINARY_DIR}")
file(GLOB SDK_LIBS "${SDK}/lib/*.so")
target_link_libraries(ioswitch_on_board ${SDK_LIBS} pthread dl rt)
target_compile_definitions(ioswitch_on_board PRIVATE RESOURCE_DIRECTORY="${BIN}/" BIN_FILE="${BIN}/IOSwitch.bin")
set_target_properties(ioswitch_on_board PROPERTIES BUILD_RPATH "${SDK}/lib")

add_executable(ioswitch_check_config check_config.cpp)
target_include_directories(ioswitch_check_config PRIVATE "${NPC_TEST}/3rd_party/nlohmann-json-develop/include/nlohmann" "${NPC_TEST}/3rd_party/nlohmann-json-develop/include")
